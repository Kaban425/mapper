=== src-tauri/src/platform/win/shell.rs -> explorer arguments
fn explorer() -> std::io::Result<PathBuf> {
    let mut buffer = [0u16; 260];
    let length = unsafe { GetSystemWindowsDirectoryW(Some(&mut buffer)) } as usize;
    Ok(PathBuf::from(String::from_utf16_lossy(&buffer[..length])).join("explorer.exe"))
}

pub fn open_folder(folder: &Path) -> std::io::Result<()> {
    Command::new(explorer()?).arg(folder).spawn()?;
    Ok(())
}

pub fn open_link(link: &str) -> std::io::Result<()> {
    Command::new(explorer()?).args(["/select,", link]).spawn()?;
    Ok(())
}
